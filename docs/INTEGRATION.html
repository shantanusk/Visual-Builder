Visual Builder — Integration Guide

How to integrate, extend, and connect the Visual Builder with your application and third-party services.

Overview

Visual Builder (VB) is a drag-and-drop page builder written in vanilla JavaScript. It runs entirely in the browser and can be embedded into any web application with minimal setup. This document covers how to integrate VB into your application, extend it with custom components, and connect it with third-party services.

Files

FilePurpose
index.htmlMain layout — toolbar, canvas, sidebar, export modal
app.jsCore logic — registry, state, rendering, DnD, export
style.cssAll styles

1. Developer API Integration

1.1 Component Registry

All components are registered in the REGISTRY object (app.js:24-508). You can add, modify, or remove components at runtime.

Reading the registry

// List all registered component types
Object.keys(REGISTRY);

// Get a specific component definition
const heroDef = REGISTRY['hero'];
console.log(heroDef.label, heroDef.category);

Adding a component dynamically

REGISTRY['myComponent'] = {
  category: 'ui',
  label: 'My Component',
  icon: '\u2605',
  defaultProps: { text: 'Hello', color: '#333' },
  template: (props) => `<div style="color:${props.color}">${props.text}</div>`,
  propsConfig: [
    { key: 'text', label: 'Text', type: 'text' },
    { key: 'color', label: 'Color', type: 'color' }
  ]
};

// Re-render the palette to show the new component
renderPalette();

Removing a component

delete REGISTRY['myComponent'];
renderPalette();

1.2 Programmatic Control

You can control the builder entirely via JavaScript.

// Add a component to the canvas
addComponent('hero');                    // appends at end
addComponent('button', 0);              // insert at index 0

// Select a component by ID
selectComponent('comp_5');

// Delete a component by ID
deleteComponent('comp_5');

// Duplicate a component by ID
duplicateComponent('comp_5');

// Update a component property
const comp = findComponentById('comp_5', canvas);
if (comp) {
  comp.props.heading = 'New Heading';
  renderCanvas();
  renderProps();
}

// Clear the entire canvas
clearCanvas();

// Undo last action
undo();

1.3 State Management

The builder state is stored in the canvas array (global).

// Snapshot the current state (save)
const snapshot = JSON.parse(JSON.stringify(canvas));

// Restore state (load)
canvas = JSON.parse(JSON.stringify(snapshot));
selectedId = null;
renderCanvas();
renderProps();

State structure

// Each canvas item:
{
  id: 'comp_1',
  type: 'hero',
  props: {
    heading: 'Build Something Amazing',
    subtitle: '...',
    ctaText: 'Get Started',
    bgColor: '#1a1a2e',
    textColor: '#ffffff',
    align: 'center'
  },
  children: [ /* nested component states */ ],
  columns: [ /* column-based containers only */ ]
}

1.4 Event Hooks

VB does not have a formal event system, but you can monkey-patch key functions to add hooks.

// Hook into component add
const _origAddComponent = addComponent;
addComponent = function(type, index) {
  _origAddComponent(type, index);
  document.dispatchEvent(new CustomEvent('vb:component-added', {
    detail: { type, id: /* last added id */ }
  }));
};

// Hook into selection changes
const _origSelect = selectComponent;
selectComponent = function(id) {
  _origSelect(id);
  document.dispatchEvent(new CustomEvent('vb:selected', {
    detail: { id }
  }));
};

// Hook into canvas changes
const _origRender = renderCanvas;
renderCanvas = function() {
  _origRender();
  document.dispatchEvent(new CustomEvent('vb:render'));
};

// Hook into export
const _origShowExport = showExport;
showExport = function() {
  _origShowExport();
};

1.5 Canvas Export & Import

Export as full HTML page

function exportFullPage() {
  const bodyHtml = canvas.map(comp => {
    const entry = REGISTRY[comp.type];
    return entry ? entry.template(comp.props) : '';
  }).join('\n\n');

  return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Built with Visual Builder</title>
</head>
<body>
${bodyHtml}
</body>
</html>`;
}

Export as HTML snippet

function exportSnippet() {
  return canvas.map(comp => {
    const entry = REGISTRY[comp.type];
    return entry ? entry.template(comp.props) : '';
  }).join('\n\n');
}

Save / load from localStorage

function saveToStorage(key = 'vb-canvas') {
  localStorage.setItem(key, JSON.stringify(canvas));
}

function loadFromStorage(key = 'vb-canvas') {
  const data = localStorage.getItem(key);
  if (data) {
    canvas = JSON.parse(data);
    selectedId = null;
    renderCanvas();
    renderProps();
  }
}

1.6 Undo / Redo

// Access undo history
console.log('Undo entries:', undoStack.length);

// Clear undo history
undoStack.length = 0;

The undoStack is a global array. Each push saves a deep clone of the canvas. Default limit: 50 entries.

2. Adding Custom Components

Components are defined in the REGISTRY object. Each entry requires:

FieldTypeDescription
categorystringOne of layout, navigation, ui, content, typography
labelstringDisplay name in palette
iconstringEmoji/text icon shown in palette
defaultPropsobjectDefault property values
templatefunctionReturns HTML; receives (props, childrenHtml, columnsHtml)
propsConfigarrayProperty editor definitions

2.1 Basic Component

REGISTRY['alert_banner'] = {
  category: 'ui',
  label: 'Alert Banner',
  icon: '\uD83D\uDD14',
  defaultProps: {
    message: 'Important notice!',
    bgColor: '#fff3cd',
    textColor: '#856404',
    dismissible: true
  },
  template: (p) => `
    <div style="
      background:${p.bgColor};
      color:${p.textColor};
      padding:12px 20px;
      border:1px solid ${p.textColor}44;
      border-radius:4px;
      margin:8px 12px;
      display:flex;
      align-items:center;
      justify-content:space-between;
      font-size:14px;
    ">
      <span>${p.message}</span>
      ${p.dismissible ? '<button style="background:none;border:none;font-size:18px;cursor:pointer;">&times;</button>' : ''}
    </div>`,
  propsConfig: [
    { key: 'message', label: 'Message', type: 'text' },
    { key: 'bgColor', label: 'Background', type: 'color' },
    { key: 'textColor', label: 'Text Color', type: 'color' },
    { key: 'dismissible', label: 'Dismissible', type: 'checkbox' }
  ]
};

2.2 Container Component (with children)

Use the children parameter in the template to support nesting.

REGISTRY['card_group'] = {
  category: 'layout',
  label: 'Card Group',
  icon: '\u229E',
  defaultProps: { gap: '16px', padding: '20px' },
  template: (p, children) => `
    <div style="
      display:grid;
      grid-template-columns:repeat(auto-fill,minmax(250px,1fr));
      gap:${p.gap};
      padding:${p.padding};
      background:#f8f9fa;
      border-radius:8px;
    ">
      ${children || '<p style="color:#999;margin:0;text-align:center;grid-column:1/-1;">Drop cards here</p>'}
    </div>`,
  propsConfig: [
    { key: 'gap', label: 'Gap', type: 'text' },
    { key: 'padding', label: 'Padding', type: 'text' }
  ]
};

2.3 Column-Based Component

Column components receive columnsHtml as the third argument — an array of HTML strings for each column.

REGISTRY['columns_3_1_2'] = {
  category: 'layout',
  label: '3-1-2 Columns',
  icon: '\u25A8',
  defaultProps: { gap: '20px' },
  template: (p, children, cols) => `
    <div style="
      display:grid;
      grid-template-columns:3fr 1fr 2fr;
      gap:${p.gap};
      padding:20px;
      background:#fff;
    ">
      <div data-col="0" style="min-height:40px;">
        ${cols && cols[0] ? cols[0] : '<div style="padding:12px;background:#f0f0f0;border-radius:4px;text-align:center;">Column 1</div>'}
      </div>
      <div data-col="1" style="min-height:40px;">
        ${cols && cols[1] ? cols[1] : '<div style="padding:12px;background:#f0f0f0;border-radius:4px;text-align:center;">Column 2</div>'}
      </div>
      <div data-col="2" style="min-height:40px;">
        ${cols && cols[2] ? cols[2] : '<div style="padding:12px;background:#f0f0f0;border-radius:4px;text-align:center;">Column 3</div>'}
      </div>
    </div>`,
  propsConfig: [
    { key: 'gap', label: 'Gap', type: 'text' }
  ]
};

3. Third-Party Service Integration

3.1 Exporting to a CMS

WordPress / Drupal / Joomla

function exportToCMS() {
  const html = exportFullPage();

  fetch('https://your-cms.example.com/api/pages', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      title: 'My Page',
      content: html,
      status: 'draft'
    })
  }).then(res => res.json())
    .then(data => console.log('CMS save response:', data));
}

Headless CMS (Contentful, Strapi, Sanity)

async function exportToHeadlessCMS(apiUrl, apiKey) {
  const state = JSON.parse(JSON.stringify(canvas));

  const response = await fetch(apiUrl, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`
    },
    body: JSON.stringify({
      data: {
        type: 'page',
        attributes: {
          title: 'VB Page',
          canvasState: state,
          renderedHtml: exportSnippet()
        }
      }
    })
  });

  return response.json();
}

3.2 Form Handlers

The Contact Form and Form components output static HTML. To make them functional:

<form action="https://formspree.io/f/YOUR_FORM_ID" method="POST">

Or integrate with a custom endpoint:

document.querySelectorAll('.comp-contact form').forEach(form => {
  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    const data = new FormData(form);
    const response = await fetch('/api/contact', {
      method: 'POST',
      body: JSON.stringify(Object.fromEntries(data)),
      headers: { 'Content-Type': 'application/json' }
    });
    if (response.ok) {
      form.innerHTML = '<p style="color:green;">Thank you!</p>';
    }
  });
});

3.3 Hosting & Deployment

Netlify / Vercel

async function deployToNetlify(siteName, accessToken) {
  const html = exportFullPage();

  const response = await fetch('https://api.netlify.com/api/v1/sites', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${accessToken}`
    },
    body: JSON.stringify({
      name: siteName,
      custom_domain: siteName,
      body: html
    })
  });

  return response.json();
}

Export as ZIP (requires JSZip)

async function exportAsZip() {
  const zip = new JSZip();
  zip.file('index.html', exportFullPage());
  zip.file('style.css', document.querySelector('link[href="style.css"]')?.outerHTML || '');
  const blob = await zip.generateAsync({ type: 'blob' });

  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'vb-export.zip';
  a.click();
  URL.revokeObjectURL(a.href);
}

3.4 Analytics

function track(event, data) {
  if (typeof gtag === 'function') {
    gtag('event', event, data);
  }
  if (typeof analytics === 'object') {
    analytics.track(event, data);
  }
}

const _origAdd = addComponent;
addComponent = function(type, index) {
  _origAdd(type, index);
  track('vb_component_added', { type, canvasLength: canvas.length });
};

const _origExport = showExport;
showExport = function() {
  _origExport();
  track('vb_export', { componentCount: canvas.length });
};

3.5 Custom Backend Persistence

Auto-save to backend

let saveTimeout;

function autoSave() {
  clearTimeout(saveTimeout);
  saveTimeout = setTimeout(() => {
    const data = JSON.stringify(canvas);
    fetch('/api/builder/save', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ canvas: data, updatedAt: new Date().toISOString() })
    });
  }, 2000);
}

const _origRender = renderCanvas;
renderCanvas = function() {
  _origRender();
  autoSave();
};

Load from backend

async function loadFromBackend(pageId) {
  const response = await fetch(`/api/builder/load/${pageId}`);
  const data = await response.json();

  if (data.canvas) {
    canvas = JSON.parse(data.canvas);
    idCounter = canvas.reduce((max, c) => {
      const num = parseInt(c.id.replace('comp_', ''), 10);
      return num > max ? num : max;
    }, 0);
    selectedId = null;
    renderCanvas();
    renderProps();
  }
}

The builder's own UI can be themed via CSS variables or by overriding style.css.

Light theme example

:root {
  --bg-dark: #ffffff;
  --bg-sidebar: #f5f5f5;
  --border-color: #e0e0e0;
  --text-color: #333333;
  --accent: #2563eb;
}

Output styling

The HTML output from export has no VB-specific classes that conflict with typical frameworks. Class names follow the pattern comp-{type} (e.g., comp-hero, comp-card, comp-button), so you can scope styles easily.
